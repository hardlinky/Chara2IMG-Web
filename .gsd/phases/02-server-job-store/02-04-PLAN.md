```ts
import { readFile } from "node:fs/promises";
import { extname, join } from "node:path";
import type { Hono } from "hono";
import { hasAdminSession, requireAdminSession, requireInvitedSession } from "../middleware/session";
import { archiveJob, getJobArchiveDir, listJobs, purgeExpiredJobs } from "../lib/jobStore";

function mimeTypeFromExt(ext: string): string {
  switch (ext.toLowerCase()) {
    case ".jpg":
    case ".jpeg":
      return "image/jpeg";
    case ".webp":
      return "image/webp";
    case ".gif":
      return "image/gif";
    default:
      return "image/png";
  }
}

export function registerPinnedImageRoutes(app: Hono): void {
  app.use("/api/pinned-images/*", requireInvitedSession);

  // Stats: count archived jobs
  app.get("/api/pinned-images/stats", async (c) => {
    const jobs = await listJobs();
    const archived = jobs.filter((j) => Boolean((j as Record<string, unknown>)["isArchived"]));
    return c.json({ ok: true, archivedJobCount: archived.length, totalBytes: 0 });
  });

  // Legacy routes — no-op stubs
  app.post("/api/pinned-images/backup", requireAdminSession, async (c) => {
    return c.json({ ok: true, message: "backup not supported in new store" });
  });

  app.post("/api/pinned-images/release", async (c) => {
    return c.json({ ok: true });
  });

  app.post("/api/pinned-images/reconcile", async (c) => {
    return c.json({ ok: true, reconciled: 0 });
  });

  app.get("/api/pinned-images/clients", requireAdminSession, async (c) => {
    return c.json({ ok: true, clients: [] });
  });

  app.post("/api/pinned-images/prune", requireAdminSession, async (c) => {
    const deleted = await purgeExpiredJobs();
    return c.json({ ok: true, deleted });
  });

  app.post("/api/pinned-images/prune-preview", requireAdminSession, async (c) => {
    return c.json({ ok: true, wouldDelete: [] });
  });

  // Archive list
  app.get("/api/pinned-images/archive", async (c) => {
    const jobs = await listJobs();
    const archived = jobs.filter((j) => Boolean((j as Record<string, unknown>)["isArchived"]));
    return c.json({ ok: true, jobs: archived });
  });

  // Archive a job by jobId (pin = archive)
  app.post("/api/pinned-images/archive", async (c) => {
    const payload = await c.req.json().catch(() => null);
    const jobId = typeof (payload as { jobId?: unknown } | null)?.jobId === "string"
      ? (payload as { jobId: string }).jobId
      : null;
    if (!jobId) {
      return c.json({ ok: false, error: "jobId required" }, 400);
    }
    const archived = await archiveJob(jobId);
    return c.json({ ok: archived, jobId });
  });

  // Batch archive — no-op (Phase 02 simplification)
  app.post("/api/pinned-images/archive-batch", async (c) => {
    return c.json({ ok: true, archived: 0 });
  });

  // Legacy admin operations — no-op stubs
  app.post("/api/pinned-images/purge-missing", requireAdminSession, async (c) => {
    return c.json({ ok: true, purged: 0 });
  });

  app.post("/api/pinned-images/rebuild-manifest", requireAdminSession, async (c) => {
    return c.json({ ok: true });
  });

  // My entries (archived jobs list — same as archive list for now)
  app.get("/api/pinned-images/my-entries", async (c) => {
    const jobs = await listJobs();
    const archived = jobs.filter((j) => Boolean((j as Record<string, unknown>)["isArchived"]));
    return c.json({ ok: true, entries: archived });
  });

  app.post("/api/pinned-images/purge-missing-for-client", async (c) => {
    return c.json({ ok: true, purged: 0 });
  });

  // Serve image file from archive directory by fileName
  // fileName format: {displayName}-{index}.{ext}  e.g. "a3f2c1b0-0.png"
  // The jobId is the first 8 chars portion — we search all archived job dirs for the file.
  app.get("/api/pinned-images/:fileName", async (c) => {
    const fileName = c.req.param("fileName");
    if (!fileName || /[/\\]/.test(fileName)) {
      return c.json({ ok: false, error: "Invalid file name" }, 400);
    }

    // Search archived job directories for this file
    const archiveBase = getJobArchiveDir();
    const jobs = await listJobs();
    const archivedJobIds = jobs
      .filter((j) => Boolean((j as Record<string, unknown>)["isArchived"]))
      .map((j) => j.jobId);

    for (const jobId of archivedJobIds) {
      const filePath = join(archiveBase, "jobs", jobId, fileName);
      try {
        const bytes = await readFile(filePath);
        const ext = extname(fileName);
        return new Response(bytes, {
          status: 200,
          headers: { "Content-Type": mimeTypeFromExt(ext) }
        });
      } catch {
        // ENOENT — not in this job dir, try next
      }
    }

    return c.json({ ok: false, error: "File not found" }, 404);
  });
}
```
